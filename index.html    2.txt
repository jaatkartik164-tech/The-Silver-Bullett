        // Load saved state from localStorage
        function loadState() {
            const saved = localStorage.getItem('ictSilverBulletChecklist');
            if (saved) {
                const items = document.querySelectorAll('.checklist-item');
                const states = JSON.parse(saved);
                items.forEach((item, index) => {
                    if (states[index]) {
                        item.classList.add('checked');
                    }
                });
            }
            updateProgress();
        }

        // Save state to localStorage
        function saveState() {
            const items = document.querySelectorAll('.checklist-item');
            const states = Array.from(items).map(item => item.classList.contains('checked'));
            localStorage.setItem('ictSilverBulletChecklist', JSON.stringify(states));
        }

        // Toggle checklist item - improved for mobile
        function toggleItem(element) {
            // Prevent double-firing on mobile
            if (element.dataset.toggling === 'true') return;
            element.dataset.toggling = 'true';
            
            element.classList.toggle('checked');
            saveState();
            updateProgress();
            
            // Add haptic feedback effect
            element.style.transform = 'scale(0.98)';
            setTimeout(() => {
                element.style.transform = '';
                element.dataset.toggling = 'false';
            }, 100);
        }

        // Add touch event listeners for better mobile support
        document.addEventListener('DOMContentLoaded', function() {
            const items = document.querySelectorAll('.checklist-item');
            items.forEach(item => {
                // Remove onclick and add proper event listeners
                item.removeAttribute('onclick');
                
                // Add both click and touch events for main checkbox
                item.addEventListener('click', function(e) {
                    // Don't toggle if clicking on sub-checkbox
                    if (e.target.closest('.item-bullets li')) {
                        return;
                    }
                    e.preventDefault();
                    toggleItem(this);
                });
                
                item.addEventListener('touchend', function(e) {
                    // Don't toggle if clicking on sub-checkbox
                    if (e.target.closest('.item-bullets li')) {
                        return;
                    }
                    e.preventDefault();
                    toggleItem(this);
                });
            });

            // Handle sub-checkbox clicks
            const subItems = document.querySelectorAll('.item-bullets li');
            subItems.forEach(subItem => {
                subItem.addEventListener('click', function(e) {
                    e.stopPropagation(); // Prevent parent toggle
                    toggleSubItem(this);
                });
                
                subItem.addEventListener('touchend', function(e) {
                    e.stopPropagation(); // Prevent parent toggle
                    e.preventDefault();
                    toggleSubItem(this);
                });
            });

            // Load sub-checkbox states
            loadSubCheckboxStates();
        });

        // Toggle sub-checkbox item
        function toggleSubItem(element) {
            if (element.dataset.toggling === 'true') return;
            element.dataset.toggling = 'true';
            
            element.classList.toggle('checked');
            saveSubCheckboxStates();
            
            setTimeout(() => {
                element.dataset.toggling = 'false';
            }, 100);
        }

        // Save sub-checkbox states
        function saveSubCheckboxStates() {
            const subItems = document.querySelectorAll('.item-bullets li');
            const states = {};
            subItems.forEach((item, index) => {
                states[index] = item.classList.contains('checked');
            });
            localStorage.setItem('ictSubCheckboxes', JSON.stringify(states));
        }

        // Load sub-checkbox states
        function loadSubCheckboxStates() {
            const saved = localStorage.getItem('ictSubCheckboxes');
            if (saved) {
                const states = JSON.parse(saved);
                const subItems = document.querySelectorAll('.item-bullets li');
                subItems.forEach((item, index) => {
                    if (states[index]) {
                        item.classList.add('checked');
                    }
                });
            }
        }

        // Update progress bar and stats
        function updateProgress() {
            const items = document.querySelectorAll('.checklist-item');
            const checkedItems = document.querySelectorAll('.checklist-item.checked');
            const total = items.length;
            const completed = checkedItems.length;
            const percentage = Math.round((completed / total) * 100);

            document.getElementById('progressFill').style.width = percentage + '%';
            document.getElementById('completedCount').textContent = completed;
            document.getElementById('totalCount').textContent = total;
            document.getElementById('percentComplete').textContent = percentage + '%';
        }

        // Reset checklist
        function resetChecklist() {
            if (confirm('🔄 Are you sure you want to reset the checklist?\n\nThis will uncheck all items and sub-items and prepare for a new trade setup.')) {
                const items = document.querySelectorAll('.checklist-item');
                items.forEach(item => {
                    item.classList.remove('checked');
                    // Add animation
                    item.style.opacity = '0.5';
                    setTimeout(() => {
                        item.style.opacity = '';
                    }, 300);
                });
                
                // Reset sub-checkboxes
                const subItems = document.querySelectorAll('.item-bullets li');
                subItems.forEach(item => {
                    item.classList.remove('checked');
                });
                
                localStorage.removeItem('ictSilverBulletChecklist');
                localStorage.removeItem('ictSubCheckboxes');
                updateProgress();
            }
        }

        // Initialize on load
        window.onload = function() {
            loadState();
        };

        // Keyboard shortcuts
        document.addEventListener('keydown', function(e) {
            // Press 'R' to reset
            if (e.key === 'r' || e.key === 'R') {
                if (e.ctrlKey || e.metaKey) {
                    e.preventDefault();
                    resetChecklist();
                }
            }
            // Press 'P' to print
            if (e.key === 'p' || e.key === 'P') {
                if (e.ctrlKey || e.metaKey) {
                    e.preventDefault();
                    window.print();
                }
            }
        });

        // Save journal entry
        function saveJournal() {
            const journalEntry = {
                id: Date.now(),
                timestamp: new Date().toISOString(),
                
                // Session Info
                date: document.getElementById('tradeDate').value,
                timeWindow: document.getElementById('timeWindow').value,
                sessionPhase: document.getElementById('sessionPhase').value,
                instrument: document.getElementById('instrument').value,
                
                // Bias & Setup
                bias: document.querySelector('input[name="bias"]:checked')?.value || '',
                liquidity: document.getElementById('liquidity').value,
                structure: document.getElementById('structure').value,
                
                // FVG Details
                fvgTimeframe: document.getElementById('fvgTimeframe').value,
                fvgQuality: document.getElementById('fvgQuality').value,
                fvgRange: document.getElementById('fvgRange').value,
                
                // Trade Execution
                entryPrice: document.getElementById('entryPrice').value,
                entryTime: document.getElementById('entryTime').value,
                stopLoss: document.getElementById('stopLoss').value,
                target: document.getElementById('target').value,
                positionSize: document.getElementById('positionSize').value,
                riskAmount: document.getElementById('riskAmount').value,
                riskPercent: document.getElementById('riskPercent').value,
                
                // Partials
                partial1: document.getElementById('partial1').value,
                partial2: document.getElementById('partial2').value,
                finalExit: document.getElementById('finalExit').value,
                breakEven: document.getElementById('breakEven').value,
                trailStop: document.getElementById('trailStop').value,
                
                // Results
                grossPL: document.getElementById('grossPL').value,
                rMultiple: document.getElementById('rMultiple').value,
                outcome: document.getElementById('outcome').value,
                holdingTime: document.getElementById('holdingTime').value,
                
                // Psychology
                emotionEntry: document.getElementById('emotionEntry').value,
                emotionExit: document.getElementById('emotionExit').value,
                adherence: document.querySelector('input[name="adherence"]:checked')?.value || '',
                
                // Notes
                whatWorked: document.getElementById('whatWorked').value,
                whatImprove: document.getElementById('whatImprove').value,
                lessons: document.getElementById('lessons').value,
                additionalNotes: document.getElementById('additionalNotes').value
            };

            // Validate required fields
            if (!journalEntry.date || !journalEntry.timeWindow || !journalEntry.instrument) {
                alert('⚠️ Please fill in at least Date, Time Window, and Instrument');
                return;
            }

            // Get existing journals
            let journals = JSON.parse(localStorage.getItem('ictJournals') || '[]');
            
            // Add new entry
            journals.unshift(journalEntry); // Add to beginning
            
            // Save to localStorage
            localStorage.setItem('ictJournals', JSON.stringify(journals));
            
            // Success feedback
            alert('✅ Journal entry saved successfully!');
            
            // Clear form
            clearJournal();
        }

                    radio.checked = false;
                });
            }
        }
